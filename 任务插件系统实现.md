### 3.2 插件加载器实现

```typescript
// src/renderer/plugins/PluginLoader.ts
import { ref, shallowRef, markRaw } from 'vue'
import type { PluginManifest, PluginInstance, PluginAPI } from './types'

export class PluginLoader implements PluginAPI {
  private plugins = ref<Map<string, PluginInstance>>(new Map())
  private pluginCache = ref<Map<string, string>>(new Map())
  
  app = {
    router: null,
    store: null,
    emit: (event: string, ...args: any[]) => {},
    on: (event: string, handler: (...args: any[]) => void) => {}
  }

  initialize(router: any, store: any) {
    this.app.router = router
    this.app.store = store
    this.app.emit = store.$emit.bind(store)
    this.app.on = store.$on.bind(store)
  }

  async install(pluginId: string): Promise<PluginInstance> {
    const manifest = await this.fetchManifest(pluginId)
    const code = await this.downloadPlugin(pluginId)
    await this.validateCode(code)
    const component = await this.compileComponent(code)
    
    const instance: PluginInstance = {
      manifest,
      component: markRaw(component),
      mounted: false,
      config: {}
    }
    
    this.plugins.value.set(pluginId, instance)
    this.pluginCache.value.set(pluginId, code)
    this.reportInstall(pluginId)
    
    this.onInstall?.(pluginId)
    return instance
  }

  private async fetchManifest(pluginId: string): Promise<PluginManifest> {
    const response = await fetch(`${import.meta.env.VITE_API_URL}/plugins/${pluginId}/manifest`)
    return response.json()
  }

  private async downloadPlugin(pluginId: string): Promise<string> {
    const response = await fetch(`${import.meta.env.VITE_API_URL}/plugins/${pluginId}/download`)
    return response.text()
  }

  private async validateCode(code: string): Promise<void> {
    const dangerousPatterns = [
      /eval\(/g,
      /new Function\(/g,
      /document\.write\(/g,
      /window\.location/g,
      /localStorage\.setItem/g
    ]
    
    for (const pattern of dangerousPatterns) {
      if (pattern.test(code)) {
        throw new Error('插件代码包含危险操作')
      }
    }
    
    if (code.length > 100 * 1024) {
      throw new Error('插件代码过大（超过100KB）')
    }
  }

  private async compileComponent(code: string): Promise<any> {
    const blob = new Blob([code], { type: 'application/javascript' })
    const url = URL.createObjectURL(blob)
    const module = await import(url)
    URL.revokeObjectURL(url)
    return module.default || module
  }

  private async reportInstall(pluginId: string): Promise<void> {
    try {
      await fetch(`${import.meta.env.VITE_API_URL}/plugins/${pluginId}/install`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' }
      })
    } catch (err) {
      console.error('Failed to report installation:', err)
    }
  }

  async uninstall(pluginId: string): Promise<void> {
    const instance = this.plugins.value.get(pluginId)
    if (instance) {
      instance.component?.unmount?.()
      this.plugins.value.delete(pluginId)
      this.pluginCache.value.delete(pluginId)
      
      await fetch(`${import.meta.env.VITE_API_URL}/plugins/${pluginId}/uninstall`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' }
      })
      
      this.onUninstall?.(pluginId)
    }
  }

  async update(pluginId: string): Promise<void> {
    await this.uninstall(pluginId)
    await this.install(pluginId)
    this.onUpdate?.(pluginId)
  }

  getPlugin(pluginId: string): PluginInstance | null {
    return this.plugins.value.get(pluginId) || null
  }

  getAllPlugins(): PluginInstance[] {
    return Array.from(this.plugins.value.values())
  }
}

export const pluginLoader = new PluginLoader()
```

### 4.1 数据库表设计

#### 4.1.1 插件表 (`plugins`)

```sql
-- 插件主表：存储所有可供用户安装的打卡任务插件元数据
CREATE TABLE plugins (
    id VARCHAR(64) PRIMARY KEY,                    -- 插件唯一标识（业务ID）
    name VARCHAR(128) NOT NULL,                    -- 插件名称（如"每日代码"）
    version VARCHAR(32) NOT NULL,                  -- 当前版本号（如"1.0.0"）
    description TEXT,                              -- 插件描述说明
    author VARCHAR(128),                           -- 作者信息
    category VARCHAR(64),                          -- 分类（编程、学习、健康、习惯等）
    downloads BIGINT DEFAULT 0,                    -- 累计下载量
    status VARCHAR(32) DEFAULT 'DRAFT'             -- 状态：DRAFT(草稿)/REVIEW(审核中)/PUBLISHED(已发布)/DISABLED(已禁用)
        CHECK (status IN ('DRAFT', 'REVIEW', 'PUBLISHED', 'DISABLED')),
    manifest JSONB NOT NULL,                       -- 插件清单JSON（包含名称、版本、依赖等元信息）
    code TEXT NOT NULL,                            -- 插件源代码（压缩后的JavaScript代码）
    hash VARCHAR(64) NOT NULL,                     -- 代码哈希值（SHA-256，用于版本校验和完整性检查）
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 创建时间
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP  -- 更新时间
);

-- 索引：按分类查询插件
CREATE INDEX idx_plugins_category ON plugins(category);
-- 索引：按下载量降序排序（热门插件）
CREATE INDEX idx_plugins_downloads ON plugins(downloads DESC);
-- 索引：按状态筛选（如只查询已发布的插件）
CREATE INDEX idx_plugins_status ON plugins(status);
```

#### 4.1.2 插件版本表 (`plugin_versions`)

```sql
-- 插件版本表：记录插件的历史版本，支持版本回滚和更新检查
CREATE TABLE plugin_versions (
    id SERIAL PRIMARY KEY,                              -- 自增主键
    plugin_id VARCHAR(64) NOT NULL REFERENCES plugins(id) ON DELETE CASCADE, -- 关联插件ID，插件删除时级联删除版本
    version VARCHAR(32) NOT NULL,                       -- 版本号（如"1.0.0", "1.1.0"）
    code TEXT NOT NULL,                                 -- 该版本的源代码
    changelog TEXT,                                     -- 更新日志（说明本次更新内容）
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP       -- 创建时间（版本发布时间）
);

-- 唯一索引：确保同一插件的同一版本只存在一条记录
CREATE UNIQUE INDEX idx_plugin_versions_unique ON plugin_versions(plugin_id, version);
```

#### 4.1.3 用户插件安装表 (`user_plugins`)

```sql
-- 用户插件安装表：记录用户与插件的安装关系
CREATE TABLE user_plugins (
    id SERIAL PRIMARY KEY,                              -- 自增主键
    user_id VARCHAR(64) NOT NULL,                       -- 用户ID（关联用户表，此处简化为字符串）
    plugin_id VARCHAR(64) NOT NULL REFERENCES plugins(id) ON DELETE CASCADE, -- 关联插件ID，插件删除时级联删除记录
    installed_version VARCHAR(32) NOT NULL,             -- 安装时的版本号（用于检查更新）
    installed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,   -- 安装时间
    enabled BOOLEAN DEFAULT true,                       -- 是否启用（卸载时设为false，而非删除记录）
    UNIQUE(user_id, plugin_id)                          -- 唯一约束：同一用户同一插件只能安装一次
);

-- 索引：按用户ID查询（获取用户安装的所有插件）
CREATE INDEX idx_user_plugins_user ON user_plugins(user_id);
-- 索引：按插件ID查询（获取哪些用户安装了该插件）
CREATE INDEX idx_user_plugins_plugin ON user_plugins(plugin_id);
```

public class PluginServiceImpl implements PluginService {
    
    private final PluginMapper pluginMapper;
    private final UserPluginMapper userPluginMapper;
    private final ObjectMapper objectMapper;
    
    private static final List<String> DANGEROUS_PATTERNS = Arrays.asList(
        "eval(", "new Function(", "document.write(", 
        "window.location", "localStorage.setItem"
    );

    @Override
    public IPage<PluginDTO> getPublishedPlugins(String category, int page, int size) {
        Page<Plugin> pageParam = new Page<>(page, size);
        IPage<Plugin> plugins;
        
        LambdaQueryWrapper<Plugin> queryWrapper = new LambdaQueryWrapper<>();
        queryWrapper.eq(Plugin::getStatus, PluginStatus.PUBLISHED);
        
        if (category != null && !category.isEmpty()) {
            queryWrapper.eq(Plugin::getCategory, category);
        }
        
        plugins = pluginMapper.selectPage(pageParam, queryWrapper);
        return plugins.convert(this::toDTO);
    }
    
    @Override
    public PluginDTO getPluginById(String id) {
        Plugin plugin = pluginMapper.selectById(id);
        return plugin != null ? toDTO(plugin) : null;
    }
    
    @Override

        Plugin plugin = pluginMapper.selectByIdAndStatus(id, PluginStatus.PUBLISHED.name());
        if (plugin == null) {
            throw new PluginNotFoundException(id);
        }
        
        try {
            return objectMapper.readValue(plugin.getManifest(), PluginManifestDTO.class);
        } catch (JsonProcessingException e) {
            throw new RuntimeException("Failed to parse manifest", e);
        }
    }
    
    @Override
    public String downloadPlugin(String id) {
        Plugin plugin = pluginMapper.selectByIdAndStatus(id, PluginStatus.PUBLISHED.name());
        if (plugin == null) {
            throw new PluginNotFoundException(id);
        }
        return plugin.getCode();
    }
    
    @Override
    @Transactional
    public PluginDTO uploadPlugin(PluginUploadDTO dto) {
        validateCode(dto.getCode());
        
        Plugin plugin = Plugin.builder()
            .id(dto.getId())
            .name(dto.getName())
            .version(dto.getVersion())
            .description(dto.getDescription())
            .author(dto.getAuthor())
            .category(dto.getCategory())
            .status(PluginStatus.REVIEW)
            .code(dto.getCode())
            .hash(computeHash(dto.getCode()))
            .manifest(buildManifest(dto))
            .build();
        
        pluginMapper.insert(plugin);
        log.info("Plugin uploaded: {}", plugin.getId());
        return toDTO(plugin);
    }
    
    @Override
    @Transactional
    public PluginDTO updatePlugin(String id, PluginUploadDTO dto) {
        Plugin plugin = pluginMapper.selectById(id);
        if (plugin == null) {
            throw new PluginNotFoundException(id);
        }
        
        validateCode(dto.getCode());
        
        plugin.setName(dto.getName());
        plugin.setVersion(dto.getVersion());
        plugin.setDescription(dto.getDescription());
        plugin.setAuthor(dto.getAuthor());
        plugin.setCategory(dto.getCategory());
        plugin.setCode(dto.getCode());
        plugin.setHash(computeHash(dto.getCode()));
        plugin.setManifest(buildManifest(dto));
        plugin.setStatus(PluginStatus.REVIEW);
        
        pluginMapper.updateById(plugin);
        log.info("Plugin updated: {}", plugin.getId());
        return toDTO(plugin);
    }
    
    @Override
    @Transactional
    public void deletePlugin(String id) {
        Plugin plugin = pluginMapper.selectById(id);
        if (plugin == null) {
            throw new PluginNotFoundException(id);
        }
        pluginMapper.deleteById(id);
        log.info("Plugin deleted: {}", id);
    }
    
    @Override
    @Transactional
    public void installPlugin(String pluginId, String userId) {
        Plugin plugin = pluginMapper.selectByIdAndStatus(pluginId, PluginStatus.PUBLISHED.name());
        if (plugin == null) {
            throw new PluginNotFoundException(pluginId);
        }
        
        UserPlugin existing = userPluginMapper.selectByUserIdAndPluginId(userId, pluginId);
        if (existing != null) {
            existing.setEnabled(true);
            userPluginMapper.updateById(existing);
        } else {
            UserPlugin userPlugin = UserPlugin.builder()
                .userId(userId)
                .pluginId(pluginId)
                .installedVersion(plugin.getVersion())
                .enabled(true)
                .build();
            userPluginMapper.insert(userPlugin);
        }
        
        pluginMapper.incrementDownloads(pluginId);
        log.info("Plugin installed: {} by user {}", pluginId, userId);
    }
    
    @Override
    @Transactional
    public void uninstallPlugin(String pluginId, String userId) {
        UserPlugin userPlugin = userPluginMapper.selectByUserIdAndPluginId(userId, pluginId);
        if (userPlugin == null) {
            throw new RuntimeException("Plugin not installed");
        }
        
        userPlugin.setEnabled(false);
        userPluginMapper.updateById(userPlugin);
        log.info("Plugin uninstalled: {} by user {}", pluginId, userId);
    }
    
    private void validateCode(String code) {
        if (code.length() > MAX_CODE_SIZE) {
            throw new PluginValidationException("插件代码过大（超过100KB）");
        }
        
        for (String pattern : DANGEROUS_PATTERNS) {
            if (code.contains(pattern)) {
                throw new PluginValidationException("插件代码包含危险操作: " + pattern);
            }
        }
    }
    
    private String computeHash(String code) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(code.getBytes(StandardCharsets.UTF_8));
            StringBuilder hexString = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) hexString.append('0');
                hexString.append(hex);
            }
            return hexString.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("SHA-256 not available", e);
        }
    }
    
    private String buildManifest(PluginUploadDTO dto) {
        PluginManifestDTO manifest = PluginManifestDTO.builder()
            .id(dto.getId())
            .name(dto.getName())
            .version(dto.getVersion())
            .description(dto.getDescription())
            .author(dto.getAuthor())
            .category(dto.getCategory())
            .dependencies(List.of())
            .entry("index.js")
            .createdAt(java.time.LocalDateTime.now().toString())
            .updatedAt(java.time.LocalDateTime.now().toString())
            .build();
        
        try {
            return objectMapper.writeValueAsString(manifest);
        } catch (JsonProcessingException e) {
            throw new RuntimeException("Failed to build manifest", e);
        }
    }
    
    private PluginDTO toDTO(Plugin plugin) {
        DateTimeFormatter formatter = DateTimeFormatter.ISO_LOCAL_DATE_TIME;
        return PluginDTO.builder()
            .id(plugin.getId())
            .name(plugin.getName())
            .version(plugin.getVersion())
            .description(plugin.getDescription())
            .author(plugin.getAuthor())
            .category(plugin.getCategory())
            .downloads(plugin.getDownloads())
            .status(plugin.getStatus().name())
            .createdAt(plugin.getCreatedAt() != null ? plugin.getCreatedAt().format(formatter) : null)
            .updatedAt(plugin.getUpdatedAt() != null ? plugin.getUpdatedAt().format(formatter) : null)
            .build();
    }
}

import org.springframework.http.HttpStatus;

import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

    public ResponseEntity<Map<String, String>> handleValidation(PluginValidationException ex) {
        Map<String, String> response = new HashMap<>();
        response.put("error", ex.getMessage());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(response);
    }
    
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, String>> handleValidationErrors(MethodArgumentNotValidException ex) {
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getAllErrors().forEach(error -> {
            String fieldName = ((FieldError) error).getField();
            String errorMessage = error.getDefaultMessage();
            errors.put(fieldName, errorMessage);
        });
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(errors);
    }
    
    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, String>> handleGenericError(Exception ex) {
        Map<String, String> response = new HashMap<>();
        response.put("error", "Internal server error");
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response);
    }
}

export class PluginSandbox {
  private allowedGlobals = [
    'console', 'setTimeout', 'clearTimeout', 
    'setInterval', 'clearInterval', 'Promise'
  ]
  
  execute(code: string): any {
    const sandbox = {} as Record<string, any>

      sandbox[key] = window[key]
    })
    
    const wrapper = `

        ${code}
      })(${this.allowedGlobals.map(g => `window['${g}']`).join(', ')});

  }

通过这种设计，用户可以动态安装、卸载和更新打卡任务插件，实现功能的灵活扩展。
